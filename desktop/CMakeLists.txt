cmake_minimum_required(VERSION 3.16)
project(sensecap_desktop LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_C_STANDARD 99)
set(ROOT "${CMAKE_CURRENT_SOURCE_DIR}/..")
option(DASH_SANITIZE "Enable address/undefined sanitizers" OFF)
if(DASH_SANITIZE)
  add_compile_options(-fsanitize=address,undefined -fno-omit-frame-pointer)
  add_link_options(-fsanitize=address,undefined)
endif()
add_library(cjson_desktop "${ROOT}/vendor/cJSON/cJSON.c")
target_include_directories(cjson_desktop PUBLIC "${ROOT}/vendor/cJSON")
add_library(dash_core "${ROOT}/main/core/model.cpp" "${ROOT}/shared/lora/protocol.cpp" "${ROOT}/shared/lora/classifier.cpp")
target_include_directories(dash_core PUBLIC "${ROOT}/main" "${ROOT}/shared")
target_link_libraries(dash_core PUBLIC cjson_desktop)
target_compile_options(dash_core PRIVATE -Wall -Wextra -Werror)
enable_testing()
add_executable(core_tests "${ROOT}/tests/core_tests.cpp")
target_link_libraries(core_tests PRIVATE dash_core)
add_test(NAME core COMMAND core_tests)
add_executable(lora_tests "${ROOT}/tests/lora_tests.cpp")
target_link_libraries(lora_tests PRIVATE dash_core)
target_compile_options(lora_tests PRIVATE -Wall -Wextra -Werror)
add_test(NAME lora COMMAND lora_tests)
add_executable(sniffer_serial_tests "${ROOT}/tests/sniffer_serial_tests.cpp"
    "${ROOT}/lora-sniffer/src/serial_diagnostics.cpp")
target_include_directories(sniffer_serial_tests PRIVATE "${ROOT}/lora-sniffer/include")
target_link_libraries(sniffer_serial_tests PRIVATE dash_core)
target_compile_options(sniffer_serial_tests PRIVATE -Wall -Wextra -Werror -UNDEBUG)
add_test(NAME sniffer_serial COMMAND sniffer_serial_tests)
find_package(Python3 COMPONENTS Interpreter REQUIRED)
add_test(NAME bridge COMMAND "${Python3_EXECUTABLE}" -m unittest discover -s "${ROOT}/bridge/tests" -v)
add_test(NAME lora_tools COMMAND "${Python3_EXECUTABLE}" "${ROOT}/tests/test_lora_tools.py" -v)
option(DASH_BUILD_SIMULATOR "Build shared-LVGL SDL simulator" ON)
if(DASH_BUILD_SIMULATOR)
  find_package(PkgConfig REQUIRED)
  pkg_check_modules(SDL2 REQUIRED IMPORTED_TARGET sdl2)
  file(GLOB_RECURSE LVGL_SOURCES CONFIGURE_DEPENDS "${ROOT}/vendor/seeed-esp32/components/lvgl/src/*.c")
  add_library(lvgl_desktop STATIC ${LVGL_SOURCES})
  target_include_directories(lvgl_desktop PUBLIC
    "${ROOT}/vendor/seeed-esp32/components/lvgl" "${CMAKE_CURRENT_SOURCE_DIR}")
  target_compile_definitions(lvgl_desktop PUBLIC LV_CONF_INCLUDE_SIMPLE=1)
  target_link_libraries(lvgl_desktop PUBLIC m)
  add_executable(simulator simulator.cpp "${ROOT}/main/ui/dashboard.cpp" "${ROOT}/main/ui/lora_page.cpp")
  target_compile_options(simulator PRIVATE -Wall -Wextra -Wpedantic)
  target_link_libraries(simulator PRIVATE dash_core lvgl_desktop PkgConfig::SDL2)
  add_test(NAME ui_simulated COMMAND simulator --simulate --self-test)
  add_test(NAME ui_unavailable COMMAND simulator --unavailable --self-test)
endif()
